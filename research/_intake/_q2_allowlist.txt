555:     summary = payload["summary"]["live_config"]
556:     rows = [row for row in payload["records"] if row.get("source_enabled") is True]
557: 
558:     kinds = Counter(str(row.get("record_kind")) for row in rows)

577:             )
578: 
579:     ids = sorted({str(row["ref_name"]) for row in primary_rows if row.get("ref_name")})
